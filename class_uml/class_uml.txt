' A "special period" is a fixed, recurring calendar period (e.g. Ramadan)
' during which the user's dietary rules and macronutrient adjustments differ
' from their baseline preferences. The nutrition plan generation algorithm
' detects whether such a period is currently active and applies its
' adjustments automatically.
class DietaryPreference {
  - dietTypes            : List<DietType>
  - allergies            : List<String>
  - likedFoods           : List<String>
  - dislikedFoods        : List<String>
  - specialPeriodName    : String
  - specialPeriodStart   : Date
  - specialPeriodEnd     : Date
  - specialAdjustments   : Map<String, String>
  --
  + addAllergy(allergy: String)                     : void
  + removeAllergy(allergy: String)                  : void
  + updateDietTypes(types: List<DietType>)          : void
  + addFoodPreference(food: String, liked: Boolean) : void
  + isSpecialPeriodActive(date: Date)               : Boolean
  + hasConflicts()                                  : Boolean
}

class MacronutrientQuota {

  - proteinG     : float

  - fatG         : float
  - fiberG       : float

  + isWithinQuota(actual: MacronutrientQuota) : Boolean
  + add(other: MacronutrientQuota)            : MacronutrientQuota
}

' ==============================================================
'  RELATIONSHIPS
' ==============================================================

' --- User & Profile ---
User           "1" *--  "1"    UserProfile        : owns >
User           "1" *--  "0..*" NutritionPlan      : owns >
User           "1" *--  "0..*" Rating             : submits >
User           "1" *--  "0..*" Report             : requests >
User           "1" *--  "0..*" Notification       : receives >
User           "1" *--  "0..1" MealPrepSchedule   : defines >

' --- UserProfile internals ---
UserProfile    "1" *--  "1"    MacronutrientQuota : has >
UserProfile    "1" *--  "1"    DietaryPreference  : has >
UserProfile    "1" *--  "0..1" HealthData         : stores >

MacronutrientQuota ..> UserProfile : <<uses>>
MacronutrientQuota ..> HealthData  : <<uses>>

' --- Meal Plan ---
NutritionPlan  "1" *--  "1..*" Meal               : contains >
NutritionPlan  "1" --o  "0..1" GroceryList        : generates >
NutritionPlan  "1" --   "0..*" ShareableLink      : produces >
NutritionPlan      ..>          MealPrepSchedule   : <<constrained by>>

' --- Meal & Ingredient (via junction) ---
Meal           "1" *--  "1..*" MealIngredient     : composed of >
MealIngredient "*" -->  "1"    Ingredient         : references >
Meal           "1" *--  "0..1" Rating             : rated via >

' --- Grocery ---
GroceryList    "1" *--  "1..*" GroceryListItem    : contains >
GroceryListItem "*" -->  "1"   Ingredient         : references >
GroceryList    "1" -->  "0..1" GroceryOrder       : leads to >
GroceryOrder   "*" -->  "1"    GroceryProvider    : placed via >
GroceryOrder   "1" *--  "1"    TrackingInfo       : tracked by >
GroceryProvider "1" *-- "1"    AuthToken          : secured by >

' --- Provider Inheritance ---
GroceryProvider <|-- DoorDashProvider
GroceryProvider <|-- KrogerProvider

' --- User ↔ GroceryProvider (many-to-many: a user can connect multiple providers) ---
User           "*" o--  "0..*" GroceryProvider    : connects to >

' --- Nutrition Info & Research ---
NutritionInfoSection "1" --> "0..*" NutritionInfoResult    : returns >
NutritionInfoSection "1" --> "0..*" AcademicSearchResult   : returns >
AIResearchAssistant  "1" *-- "0..*" Message                : history >
AIResearchAssistant  "1" --> "0..*" AssistantResponse      : produces >
AssistantResponse    "*" o-- "0..*" AcademicSearchResult   : cites >
